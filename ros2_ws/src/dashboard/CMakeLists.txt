cmake_minimum_required(VERSION 3.8)
project(uamc_dashboard)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
find_package(uamc_interfaces REQUIRED)
find_package(ackermann_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
add_library(${PROJECT_NAME} SHARED src/dashboard_panel.cpp include/uamc_dashboard/dashboard_panel.hpp)
target_include_directories(${PROJECT_NAME} PUBLIC include)
ament_target_dependencies(${PROJECT_NAME} rviz_common pluginlib rclcpp rclcpp_action uamc_interfaces ackermann_msgs nav_msgs sensor_msgs std_srvs)
target_link_libraries(${PROJECT_NAME} Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins_description.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY config/ DESTINATION share/${PROJECT_NAME}/config)
if(BUILD_TESTING)
  add_executable(dashboard_gui_smoke test/dashboard_gui_smoke.cpp)
  ament_target_dependencies(dashboard_gui_smoke rviz_common rviz_rendering rclcpp)
  target_link_libraries(dashboard_gui_smoke Qt5::Widgets)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_dashboard_panel test/test_dashboard_panel.cpp)
  target_include_directories(test_dashboard_panel PRIVATE include)
  ament_target_dependencies(test_dashboard_panel rviz_common rclcpp)
  target_link_libraries(test_dashboard_panel ${PROJECT_NAME} Qt5::Widgets)
  set_tests_properties(test_dashboard_panel PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  ament_add_gtest(test_dashboard_state test/test_dashboard_state.cpp)
  target_include_directories(test_dashboard_state PRIVATE include)
  ament_add_gtest(test_ros_client test/test_ros_client.cpp)
  target_include_directories(test_ros_client PRIVATE include)
  ament_target_dependencies(test_ros_client rviz_common pluginlib rclcpp rclcpp_action uamc_interfaces ackermann_msgs nav_msgs sensor_msgs std_srvs)
  target_link_libraries(test_ros_client Qt5::Widgets)
endif()
ament_package()
